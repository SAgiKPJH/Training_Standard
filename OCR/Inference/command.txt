#테스트시
# 1. 기존 venv 삭제
Remove-Item -Recurse -Force .venv

# 2. Python 3.10로 재생성 (설치되어 있어야 함)
py -3.10 -m venv .venv

# 3. paddle GPU 먼저 설치
.venv\Scripts\pip install paddlepaddle-gpu==2.6.0 -f https://www.paddlepaddle.org.cn/whl/windows/mkl/avx/stable.html

# 4. 나머지 패키지
.venv\Scripts\pip install -r requirements.txt

# 5. 사전학습 모델 다운로드 (인터넷 되는 PC에서 실행)
#    -> data/det_model/, data/cls_model/ 생성됨, 내부망 PC로 복사
.venv\Scripts\python download.py

# 6. 실행
.venv\Scripts\python model_handler.py
